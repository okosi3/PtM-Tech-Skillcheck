/// Script for tavern button perception check
/// 	Button id: baad8590-056a-4e12-9d9d-201d15051a21
/// 	BoxTrigger id: 294c4e93-c08f-4d4a-934f-9118bd022286

//REGION RevealButton

// Perception check if player enters boxtrigger
IF 
EnteredTrigger(_Player, (TRIGGER)Tavern_Button_BoxTrigger_294c4e93-c08f-4d4a-934f-9118bd022286)
AND
DB_Players(_Player)
THEN
SetCanInteract((ITEM)Tavern_Door_Button_baad8590-056a-4e12-9d9d-201d15051a21, 1);
RequestPassiveRoll(_Player, Tavern_Door_Button_baad8590-056a-4e12-9d9d-201d15051a21, "Passive", "Perception", (DIFFICULTYCLASS)HiddenPerception_Trivial_7d16b9fd-d3c3-4693-9d50-ae44b962b4c4, 0, "PTM_TavernButtonPerceptionCheck");

// Result handling
PROC 
PROC_RollResult("PTM_TavernButtonPerceptionCheck", _Player, Tavern_Door_Button_baad8590-056a-4e12-9d9d-201d15051a21, 1)
THEN 
SetCanInteract((ITEM)Tavern_Door_Button_baad8590-056a-4e12-9d9d-201d15051a21, 1);
PROC_PlayPerceptionRevealEffect(Tavern_Door_Button_baad8590-056a-4e12-9d9d-201d15051a21, "PTM_TavernButtonPerceptionCheck");

//END_REGION
