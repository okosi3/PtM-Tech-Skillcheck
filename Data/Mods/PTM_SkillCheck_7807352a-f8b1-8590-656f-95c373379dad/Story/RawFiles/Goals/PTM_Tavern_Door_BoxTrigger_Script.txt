/// Script for tavern button perception check
/// 	Button id: baad8590-056a-4e12-9d9d-201d15051a21
/// 	BoxTrigger id: 294c4e93-c08f-4d4a-934f-9118bd022286

// Perception check if player enters boxtrigger
IF
    CharacterEnteredTrigger(_Character, "Tavern_Button_BoxTrigger_294c4e93-c08f-4d4a-934f-9118bd022286")
    AND
    IsPlayer(_Character)
    AND
    NOT DB_TavernButton_CheckDone(_Character) // Prevents future triggers
THEN
    DB_TavernButton_CheckDone(_Character);
    PROC_TrySkillCheck(
        _Character,
        "Tavern_Button_BoxTrigger_294c4e93-c08f-4d4a-934f-9118bd022286",
        "Perception",
        10,
        "PTM_TavernButtonPerceptionCheck"
    );


// Result handling
PROC
	PROC_RollResult("PTM_TavernButtonPerceptionCheck", _Character, "Tavern_Button_BoxTrigger_294c4e93-c08f-4d4a-934f-9118bd022286", 1)
THEN
	DB_TavernButton_Revealed();
    SetCanInteract("Tavern_Door_Button_baad8590-056a-4e12-9d9d-201d15051a21", 1);

//END_REGION
EXITSECTION

ENDEXITSECTION
